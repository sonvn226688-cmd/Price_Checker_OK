const $ = id => document.getElementById(id);

function normalize(value) {
  return String(value || "")
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g,"")
    .replace(/đ/g,"d").replace(/Đ/g,"D")
    .toLowerCase().replace(/[^a-z0-9]/g,"");
}

function getField(row, aliases) {
  const keys = Object.keys(row);
  for (const alias of aliases) {
    const key = keys.find(k => normalize(k) === normalize(alias));
    if (key && row[key] !== "") return row[key];
  }
  return "";
}

const fields = {
  exporter:["Exporter","Export Company","Seller","Shipper",
    "Cong ty xuat khau","Ten cong ty xuat khau","Export name"],
  importer:["Importer","Import Company","Buyer","Consignee",
    "Cong ty nhap khau","Ten cong ty nhap khau","Import name"],
  product:["Product","Commodity","Goods","Description",
    "Hang hoa","Loai hang","Ten hang"],
  price:["Price","Unit Price","Value","Gia","Don gia"],
  currency:["Currency","Tien te","Unit"],
  transport:["Transport","Shipping","Shipment","Mode",
    "Van chuyen","Phuong thuc van chuyen"],
  term:["Incoterm","Trade Term","Terms","Term",
    "Dieu kien giao hang"],
  date:["Date","Shipment Date","Ngay giao dich"],
  quantity:["Quantity","Qty","So luong"],
  origin:["Origin","Country of Origin","Xuat xu"],
  destination:["Destination","Country","Diem den"]
};

function value(row, field) {
  return getField(row, fields[field] || []);
}

function escapeHTML(str) {
  return String(str ?? "").replace(/[&<>"']/g, c => ({
    "&":"&amp;","<":"&lt;",">":"&gt;",
    '"':"&quot;","'":"&#39;"
  })[c]);
}

function jsonp(url) {
  return new Promise((resolve,reject) => {
    const callback = "tradeCallback" + Date.now();
    const script = document.createElement("script");
    const timer = setTimeout(() => {
      cleanup();
      reject(new Error("Hết thời gian kết nối"));
    },15000);

    function cleanup() {
      clearTimeout(timer);
      delete window[callback];
      script.remove();
    }

    window[callback] = data => {
      cleanup();
      resolve(data);
    };
    script.onerror = () => {
      cleanup();
      reject(new Error("Không thể đọc Google Sheets"));
    };
    script.src = url + (url.includes("?") ? "&" : "?")
      + "callback=" + callback + "&t=" + Date.now();
    document.body.appendChild(script);
  });
}

async function loadData() {
  if (API_URL.includes("YOUR_GOOGLE")) {
    $("status").textContent = "Chưa cấu hình Google Sheets API.";
    return;
  }
  try {
    $("status").textContent = "Đang tải dữ liệu...";
    const data = await jsonp(API_URL);
    if (!data.success) throw new Error(data.error || "Lỗi dữ liệu");
    database = data.rows || [];
    $("status").textContent =
      "Đã tải " + database.length + " giao dịch thành công.";
  } catch(error) {
    $("status").textContent = "Lỗi: " + error.message;
  }
}

function findCompanies(query) {
  const q = normalize(query);
  if (!q) return [];

  const names = [...new Set(database
    .map(r => String(value(r,"exporter") || "").trim())
    .filter(Boolean))];

  return names.map(name => {
    const n = normalize(name);
    let score = n === q ? 100 :
      n.startsWith(q) ? 80 :
      n.includes(q) ? 60 : 0;

    if (!score && q.length >= 4 && n.includes(q.slice(0,4)))
      score = 20;

    return {name,score};
  }).filter(x => x.score > 0)
    .sort((a,b) => b.score-a.score)
    .slice(0,15);
}

function showSuggestions() {
  const matches = findCompanies($("search").value);
  const box = $("suggestions");

  if (!matches.length) {
    box.classList.add("hidden");
    return;
  }

  box.innerHTML = matches.map(x =>
    `<div class="suggestion" data-name="${escapeHTML(x.name)}">
      ${escapeHTML(x.name)}
    </div>`
  ).join("");

  box.classList.remove("hidden");
  box.querySelectorAll(".suggestion").forEach(el => {
    el.onclick = () => {
      $("search").value = el.dataset.name;
      box.classList.add("hidden");
      search();
    };
  });
}

  const rows = database.filter(r =>
    normalize(value(r,"exporter")) === normalize(name)
  );

  const importers = [...new Set(rows.map(r =>
    String(value(r,"importer") || "").trim()
  ).filter(Boolean))];

  const details = rows.map((r,i) => `
    <article class="company">
      <div class="company-name">${i+1}. ${escapeHTML(value(r,"product") || "Chưa có dữ liệu hàng hóa")}</div>
      <div class="meta">Công ty nhập khẩu: ${escapeHTML(value(r,"importer") || "Chưa cập nhật")}</div>
      <table>
        <tr><td>Giá giao dịch</td><td>${escapeHTML(value(r,"price") || "—")} ${escapeHTML(value(r,"currency"))}</td></tr>
        <tr><td>Vận chuyển</td><td>${escapeHTML(value(r,"transport") || "—")}</td></tr>
        <tr><td>Incoterms</td><td>${escapeHTML(value(r,"term") || "—")}</td></tr>
        <tr><td>Số lượng</td><td>${escapeHTML(value(r,"quantity") || "—")}</td></tr>
        <tr><td>Xuất xứ</td><td>${escapeHTML(value(r,"origin") || "—")}</td></tr>
        <tr><td>Điểm đến</td><td>${escapeHTML(value(r,"destination") || "—")}</td></tr>
        <tr><td>Ngày giao dịch</td><td>${escapeHTML(value(r,"date") || "—")}</td></tr>
      </table>
    </article>
  `).join("");

  $("results").innerHTML = `
    <article class="company">
      <div class="company-name">${escapeHTML(name)}</div>
      <div class="meta">${rows.length} giao dịch được tìm thấy</div>
      <span class="pill">${importers.length} CÔNG TY NHẬP KHẨU</span>
      <table>
        <tr><td>Danh sách người mua</td>
        <td>${importers.length ? importers.map(escapeHTML).join("<br>") : "Chưa có dữ liệu"}</td></tr>
      </table>
    </article>
    <div class="section-title">CHI TIẾT GIAO DỊCH (${rows.length})</div>
    ${details}
  `;
}

function search() {
  const q = $("search").value.trim();
  $("suggestions").classList.add("hidden");

  if (!q) {
    $("results").innerHTML =
      '<div class="empty">Vui lòng nhập tên công ty.</div>';
    return;
  }

  const matches = findCompanies(q);
  if (!matches.length) {
    $("results").innerHTML =
      '<div class="empty">Không tìm thấy công ty phù hợp.</div>';
    return;
  }

  // Tìm gần đúng nhưng chỉ hiển thị công ty khớp nhất
  renderCompany(matches[0].name);
}
